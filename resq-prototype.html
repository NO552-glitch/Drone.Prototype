# resq_core/mission_loop.py
while mission.active:
    frame_rgb, frame_ir = sensors.capture()
    dets = yolo_fusion(frame_rgb, frame_ir)

    if gps.signal_lost() or env.is_collapsed_structure:
        pose = vio_lidar_slam.update(imu, lidar)
        nav.mode = "SLAM_GPS_DENIED"
    else:
        pose = gps.get_pose()
        nav.mode = "GPS_NAV"

    for d in dets:
        risk = hazard_risk(d, env.map)
        priority = 0.6 * d.confidence + 0.4 * risk
        rescue_queue.push(d, priority, pose)

    heatmap.mark_cell(pose.cell, covered=True)
    nav.step_waypoint(coverage_plan)

    if comms.link_available():
        sync.flush(rescue_queue, heatmap)
    else:
        store.buffer_local(rescue_queue, heatmap)
